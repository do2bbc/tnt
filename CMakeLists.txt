cmake_minimum_required(VERSION 3.20)
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "Release C compiler flags")
project(tnt VERSION 2.0.0 LANGUAGES C)
set(TNT_VERSION "2.0.0-rc3")

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use a separate build directory: cmake -S . -B build")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)

include(CheckStructHasMember)

option(TNT_SERIAL "Enable serial and soft-TNC device handling" ON)
option(TNT_SOUND "Enable sound support" ON)
option(TNT_NLS "Build system-gettext translations" ON)
option(TNT_AX25K "Enable legacy Linux AX.25 library support" OFF)
option(TNT_AX25K2 "Enable modern Linux AX.25 library support" OFF)
option(TNT_HIBAUD "Enable Linux high-baud-rate support" OFF)
option(TNT_GENUSER "Enable legacy Unix user generation" OFF)
option(TNT_DPBOXT "Enable the unsupported historical DPBOXT mode" OFF)
option(TNT_FLAT "Install everything under PREFIX/tnt" OFF)
option(TNT_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
set(TNT_LANGUAGE en CACHE STRING "Default configuration language: en, de or fr")
set_property(CACHE TNT_LANGUAGE PROPERTY STRINGS en de fr)
if(NOT TNT_LANGUAGE MATCHES "^(en|de|fr)$")
  message(FATAL_ERROR "TNT_LANGUAGE must be en, de or fr")
endif()
if(TNT_DPBOXT)
  message(FATAL_ERROR "DPBOXT was already unsupported in the Autotools build; use the standard TNT terminal")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND
    (TNT_AX25K OR TNT_AX25K2 OR TNT_HIBAUD))
  message(FATAL_ERROR "AX.25 kernel and HIBAUD options require Linux")
endif()

# Preserve TNT's historical /var default, including user-prefix installations.
set(TNT_LOCALSTATEDIR /var CACHE PATH "Runtime state root (previously --localstatedir)")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(TNT_PLATFORM TNT_LINUX)
  set(cfg_file linux)
  set(DIR_STRING "ls -xaFT80 ")
  set(DIRL_STRING "ls -laFT80 ")
  set(DIRRUN_STRING "ls -xT80 ")
elseif(CMAKE_SYSTEM_NAME MATCHES "^(NetBSD|OpenBSD)$")
  set(TNT_PLATFORM TNT_NETBSD)
  set(cfg_file netbsd)
  set(DIR_STRING "ls -la ")
  set(DIRL_STRING "ls -la ")
  set(DIRRUN_STRING "ls ")
elseif(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
  set(TNT_PLATFORM TNT_SOLARIS)
  set(cfg_file solaris)
  set(DIR_STRING "ls -lF ")
  set(DIRL_STRING "ls -lF ")
  set(DIRRUN_STRING "ls ")
  set(SPECIAL 1)
  set(PWD_NOT_EMPTY 1)
else()
  message(FATAL_ERROR "Unsupported platform: ${CMAKE_SYSTEM_NAME}")
endif()
set(FREE_STRING "df 2>/dev/null > ")

  set(DEF_AUTO_NEWLINE 1)
  set(DEF_SUPP_HICNTL 1)
  set(DEF_TERMCAP 1)
endif()

  set(tnt_doc_dir "${CMAKE_INSTALL_PREFIX}/share/doc/tnt")

  set(tnt_main_dir "${CMAKE_INSTALL_PREFIX}/tnt")

  set(tnt_conf_dir "${tnt_main_dir}")

set(tnt_sound_dir "${CMAKE_INSTALL_PREFIX}/share/tnt/sounds")

  set(tnt_log_dir "${tnt_work_dir}")
  set(tnt_proc_dir "${tnt_main_dir}")
  set(tnt_sound_dir "${tnt_work_dir}/sounds")
endif()

  set(tnt_main_dir "${CMAKE_INSTALL_PREFIX}/tnt")
  foreach(kind work conf log proc)
    set(tnt_${kind}_dir "${tnt_main_dir}")
  endforeach()
  set(tnt_doc_dir "${tnt_main_dir}/documentation")
  set(tnt_bin_dir "${tnt_main_dir}/bin")
  set(tnt_sound_dir "${tnt_main_dir}/sounds")
  foreach(kind BINDIR SBINDIR)
    set(CMAKE_INSTALL_FULL_${kind} "${tnt_main_dir}")
  endforeach()
  set(CMAKE_INSTALL_FULL_INCLUDEDIR "${tnt_main_dir}/include")
  set(CMAKE_INSTALL_FULL_INFODIR "${tnt_main_dir}/documentation")
  set(CMAKE_INSTALL_FULL_LOCALEDIR "${tnt_main_dir}/locale")
endif()

set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
foreach(header termcap.h langinfo.h sys/select.h sys/wait.h termio.h)
  string(TOUPPER "${header}" variable)
  string(REGEX REPLACE "[/.]" "_" variable "${variable}")
  check_include_file("${header}" HAVE_${variable})
endforeach()
if(NOT HAVE_TERMCAP_H OR NOT HAVE_LANGINFO_H)
  message(FATAL_ERROR "termcap.h and langinfo.h are required; install ncurses development headers")
endif()
find_library(TNT_TERMCAP_LIBRARY NAMES tinfo ncursesw ncurses REQUIRED)
set(CMAKE_REQUIRED_LIBRARIES "${TNT_TERMCAP_LIBRARY}")
check_symbol_exists(tgetent termcap.h HAVE_TGETENT)
unset(CMAKE_REQUIRED_LIBRARIES)
if(NOT HAVE_TGETENT)
  message(FATAL_ERROR "The terminal library must provide tgetent")
endif()
foreach(symbol memmove index)
  string(TOUPPER "${symbol}" variable)
  check_symbol_exists("${symbol}" "string.h;strings.h" HAVE_${variable})
endforeach()
check_symbol_exists(socket sys/socket.h HAVE_SOCKET)
set(CMAKE_EXTRA_INCLUDE_FILES sys/socket.h)
check_type_size(socklen_t TNT_SOCKLEN_T)
set(HAVE_SOCKLEN_T "${HAVE_TNT_SOCKLEN_T}")
unset(CMAKE_EXTRA_INCLUDE_FILES)
check_symbol_exists(usleep unistd.h HAVE_USLEEP)
check_symbol_exists(srandom stdlib.h HAVE_SRANDOM)
check_symbol_exists(CRTSCTS termios.h HAVE_CRTSCTS)
check_struct_has_member("struct utmp" ut_host utmp.h HAVE_UT_HOST)
check_struct_has_member("struct utmp" ut_exit utmp.h HAVE_UT_EXIT)
find_program(TNT_EDITOR NAMES cmdtool editor vim vi ed)
if(NOT TNT_EDITOR)
  message(WARNING "No editor found; using the historical runtime default 'ed'. Set TNT_EDITOR to choose another editor.")
  set(TNT_EDITOR ed CACHE STRING "Default runtime editor" FORCE)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
  list(APPEND TNT_LIBRARIES nsl socket)
endif()
if(TNT_NLS)
  find_package(Intl REQUIRED)
  find_package(Gettext REQUIRED)
  set(CMAKE_REQUIRED_LIBRARIES "${Intl_LIBRARIES}")
  set(CMAKE_REQUIRED_INCLUDES "${Intl_INCLUDE_DIRS}")
  check_symbol_exists(bind_textdomain_codeset libintl.h HAVE_BIND_TEXTDOMAIN_CODESET)
  unset(CMAKE_REQUIRED_LIBRARIES)
  unset(CMAKE_REQUIRED_INCLUDES)
  if(NOT HAVE_BIND_TEXTDOMAIN_CODESET)
    message(FATAL_ERROR "System gettext must support bind_textdomain_codeset")
  endif()
  set(ENABLE_NLS 1)
  list(APPEND TNT_LIBRARIES ${Intl_LIBRARIES})
endif()
if(TNT_AX25K OR TNT_AX25K2)
  find_library(TNT_AX25_LIBRARY NAMES ax25 REQUIRED)
  set(USE_AX25K 1)
  set(USE_AX25K2 "${TNT_AX25K2}")
  foreach(header netax25/ax25.h net/ax25.h netrom/netrom.h netrose/rose.h netax25/procutils.h)
    string(TOUPPER "${header}" variable)
    string(REGEX REPLACE "[/.]" "_" variable "${variable}")
    check_include_file("${header}" HAVE_${variable})
  endforeach()
  set(CMAKE_REQUIRED_LIBRARIES "${TNT_AX25_LIBRARY}")
  if(TNT_AX25K2)
    check_symbol_exists(ax25_config_load_ports "netax25/ax25.h;netax25/axconfig.h" HAVE_AX25_CONFIG_MODERN)
    set(HAVE_AX25_CONFIG "${HAVE_AX25_CONFIG_MODERN}")
  else()
    check_symbol_exists(ax25_config_get_addr ax25/axutils.h HAVE_AX25_CONFIG_LEGACY)
    set(HAVE_AX25_CONFIG "${HAVE_AX25_CONFIG_LEGACY}")
  endif()
  unset(CMAKE_REQUIRED_LIBRARIES)
  if(NOT HAVE_AX25_CONFIG)
    message(FATAL_ERROR "Missing AX.25 headers or required library functions")
  endif()
  list(APPEND TNT_LIBRARIES "${TNT_AX25_LIBRARY}")
endif()

  "LOCALEDIR=\"${CMAKE_INSTALL_FULL_LOCALEDIR}\"")

  target_include_directories(tnt_build_options INTERFACE ${Intl_INCLUDE_DIRS})

foreach(flag -funsigned-char -Wall -Wextra -Wstrict-prototypes -Wold-style-definition)
  string(MAKE_C_IDENTIFIER "${flag}" variable)
  check_c_compiler_flag("${flag}" TNT_FLAG_${variable})
  if(TNT_FLAG_${variable})
    target_compile_options(tnt_build_options INTERFACE "${flag}")
  endif()
endforeach()
if(TNT_WARNINGS_AS_ERRORS)
  target_compile_options(tnt_build_options INTERFACE -Werror)
endif()

  target_link_libraries(${target} PRIVATE tnt_build_options ${TNT_LIBRARIES})
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/src")

foreach(target help logcall tnt-info)
  set(source "${target}")
  if(target STREQUAL "tnt-info")
    set(source info)
  endif()
  set(output "${target}")
  if(target STREQUAL "help")
    set(output tnt-help)
  endif()
  add_executable(tnt_${source} "src/runprgs/${source}.c")
  target_include_directories(tnt_${source} BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/src/runprgs")
  target_link_libraries(tnt_${source} PRIVATE tnt_build_options)
  set_target_properties(tnt_${source} PROPERTIES OUTPUT_NAME "${output}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/src/runprgs")
endforeach()

    list(APPEND catalogs "${CMAKE_CURRENT_BINARY_DIR}/po/${lang}.mo")

      DESTINATION "${CMAKE_INSTALL_FULL_LOCALEDIR}/${lang}/LC_MESSAGES" RENAME tnt.mo)
  endforeach()
  add_custom_target(translations ALL DEPENDS ${catalogs})
endif()
